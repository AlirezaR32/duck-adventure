     1→/**
     2→ * Project Duck — game orchestrator (V0.6, multi-level).
     3→ *
     4→ * Fixed-timestep simulation (1/60) with an accumulator. The duck is a
     5→ * kinematic body: gravity, jump with anticipation, coyote time and jump
     6→ * buffering. Swinging is the star — grab, pump, release, fly, land:
     7→ *
     8→ *  - every swing is independent (own pivot, length, damping, grab range)
     9→ *  - release preserves the pendulum's tangential momentum
    10→ *  - a short re-grab cooldown on the swing you just left prevents accidental
    11→ *    re-attaches, while any OTHER swing in range can be grabbed instantly
    12→ *  - explicit checkpoint posts activate as you pass them; respawning resets
    13→ *    physics, swing attachments and input safely (never mid-air stale state)
    14→ *  - moving platforms carry the duck via the physics carrier (deterministic)
    15→ *
    16→ * V0.6 adds a reusable level system: levels are data (levels.ts) built by
    17→ * buildLevelRuntime; loadLevel() tears the previous level down completely
    18→ * (scene objects, physics boxes, swings, progression) so repeated switches
    19→ * and restarts can never leak stale state. One Game instance serves the
    20→ * whole session — input handlers and the RAF loop are never duplicated.
    21→ */

    54→function moveToward(v: number, target: number, maxDelta: number): number {
    55→  return v < target ? Math.min(v + maxDelta, target) : Math.max(v - maxDelta, target)
    56→}
    57→
    58→function clamp(v: number, lo: number, hi: number): number {
    59→  return Math.max(lo, Math.min(hi, v))
    60→}

   244→  /** Tear the current level down completely, then build level `idx` fresh. */
   245→  loadLevel(idx: number): void {
   246→    const i = Math.max(0, Math.min(LEVELS.length - 1, Math.floor(idx)))
   247→    this.levelIdx = i
   248→    const def = LEVELS[i]
   249→    this.def = def
   250→
   251→    // ---- teardown: scene objects, swings, physics ----
   252→    if (this.level) this.level.dispose()
   253→    for (const s of this.swings) {
   254→      this.scene.remove(s.group)
   255→      s.dispose()
   256→    }
   257→    this.swings = []
   258→    this.world.reset()
   259→
   260→    // ---- build ----
   261→    this.level = buildLevelRuntime(this.scene, this.world, def)
   262→    for (const cfg of def.swings) {
   263→      const s = new Swing(cfg)
   264→      this.swings.push(s)
   265→      this.scene.add(s.group)
   266→    }
   267→    this.moverBoxes = this.world.boxes.filter((b) => this.world.motionOf(b) !== undefined)
   268→
   269→    // ---- per-level play state (no stale forces, attachments or counters) ----
   270→    this.plats = def.platforms
   271→    this.checkpoints = def.checkpoints
   272→    this.flagX = def.flagX
   273→    this.spawnX = def.spawnX
   274→    this.killY = def.killY
   275→    this.levelTime = 0
   276→    this.cpIndex = -1
   277→    this.starsGot = 0
   278→    this.tipSent = { pump: false, star: false }
   279→    this.clearConfetti()
   280→    this.won = false
   281→    this.hopTimer = 0
   282→    this.swinging = false
   283→    this.activeSwing = null
   284→    this.regrabSwing = null

   286→    this.whooshT = 0
   287→    this.debugRelease = null

   316→  // ============================ main loop ============================
   317→
   318→  private tick = (now: number): void => {
   319→    if (this.disposed) return
   320→    this.raf = requestAnimationFrame(this.tick)
   321→    let dt = (now - this.lastT) / 1000
   322→    this.lastT = now
   323→    dt = Math.min(dt, 0.1)
   324→    this.acc += dt
   325→    let steps = 0
   326→    while (this.acc >= T.fixedDt && steps < 5) {
   327→      this.step(T.fixedDt)
   328→      this.acc -= T.fixedDt
   329→      steps++
   330→    }
   331→    if (steps === 5) this.acc = 0
   332→    this.bg.update(dt, this.camX)
   333→    this.level.update(this.time)
   334→    for (const s of this.swings) s.tickVisual(dt, this.time)
   335→    this.tickConfetti(dt)
   336→    this.renderer.render(this.scene, this.camera)
   337→  }
   338→
   339→  private step(dt: number): void {
   340→    this.time += dt
   341→    // moving platforms advance once per fixed step, before anything moves;
   342→    // a grounded duck then rides its carrier by the platform's exact delta
   343→    this.world.updateMotions(this.time)
   344→    const b = this.body
   345→    if (b.onGround && b.carrier) {
   346→      const m = this.world.motionOf(b.carrier)
   347→      if (m) {
   348→        b.x += m.dx
   349→        b.y += m.dy
   350→      }
   351→    }
   352→    if (!this.won) this.levelTime += dt
   353→    const interact = this.input.consumeInteract()
   354→    const jump = this.input.consumeJump()
   355→    this.regrabT = Math.max(0, this.regrabT - dt)
   356→    if (this.regrabT === 0) this.regrabSwing = null
   357→
   358→    if (this.won) {
   359→      // celebratory hops at the flag
   360→      this.hopTimer -= dt
   361→      if (b.onGround && this.hopTimer <= 0) {
   362→        b.vy = 7.2
   363→        b.onGround = false
   364→        this.duck.onJump()
   365→        this.hopTimer = 0.55
   366→      }
   367→      b.vx = moveToward(b.vx, 0, T.groundDecel * dt)
   368→      const was = b.onGround
   369→      b.vy = Math.max(T.maxFall, b.vy + T.gravity * dt)
   370→      this.world.move(b, dt, was)
   371→      for (const s of this.swings) s.settle(dt, this.time)
   372→      this.syncDuck(dt)
   373→      this.updateCamera(dt)
   374→      return
   375→    }
   376→
   377→    if (this.swinging && this.activeSwing) {
   378→      const sw = this.activeSwing
   379→      // pumping: pushing in the direction of motion adds energy (leg pumping)
   380→      const dir = this.input.dir()
   381→      const align = dir !== 0 ? (Math.sign(sw.omega) === dir ? 1 : 0.25) : 0
   382→      const pump = dir * align * T.swingPump
   383→      sw.update(dt, pump, 30)
   384→      for (const s of this.swings) if (s !== sw) s.settle(dt, this.time)
   385→      // the ropes creak softly when the duck works the swing hard or sweeps
   386→      // near the top of the arc — quiet, occasional, part of the world
   387→      this.creakT -= dt
   388→      if ((Math.abs(pump) > 2.2 || Math.abs(sw.theta) > 1.02) && this.creakT <= 0) {
   389→        this.sfx.creak()
   390→        this.creakT = 0.5 + Math.random() * 0.4
   391→      }
   392→      if (this.debugRelease) {
   393→        const d = this.debugRelease
   394→        if (sw.omega > d.minOmega && sw.theta > d.lo && sw.theta < d.hi) {
   395→          this.debugRelease = null
   396→          this.release()
   397→        }
   398→      }
   399→      const a = sw.seatAnchor()
   400→      b.x = a.x
   401→      b.y = a.y + 0.08 // feet rest on the seat board
   402→      // track tangential velocity so release preserves momentum
   403→      b.vx = sw.omega * sw.L * Math.cos(sw.theta)
   404→      b.vy = sw.omega * sw.L * Math.sin(sw.theta)
   405→      this.setPrompt('LET GO')
   406→      if (jump || interact) this.release()
   407→      // airy whoosh when sweeping through the bottom of the arc at speed
   408→      this.whooshT -= dt
   409→      if (Math.abs(sw.omega) > 3.1 && Math.abs(sw.theta) < 0.24 && this.whooshT <= 0) {
   410→        this.sfx.whoosh()
   411→        this.whooshT = 0.6
   412→      }
   413→      this.collectStars()
   414→      this.syncDuck(dt, pump / T.swingPump)
   415→      this.updateCamera(dt)
   416→      return
   417→    }
   418→
   419→    // ---- horizontal movement ----
   420→    const dir = this.input.dir()
   421→    if (dir !== 0) {
   422→      const target = dir * T.moveSpeed
   423→      const sameDir = Math.sign(b.vx) === dir
   424→      const over = Math.abs(b.vx) > T.moveSpeed + 0.05
   425→      if (over && sameDir) {
   426→        // preserve hard-earned momentum (e.g. swing releases); settle
   427→        // gently back to run speed only once grounded
   428→        if (b.onGround) b.vx = moveToward(b.vx, target, 8 * dt)
   429→      } else {
   430→        const rate = b.onGround ? T.groundAccel : T.airAccel
   431→        b.vx = moveToward(b.vx, target, rate * dt)
   432→      }
   433→    } else {
   434→      b.vx = moveToward(b.vx, 0, (b.onGround ? T.groundDecel : T.airDrag) * dt)
   435→    }
   436→
   437→    // unattended swings keep settling/swaying between grabs — released
   438→    // swings must actually lose energy, otherwise they spin forever and
   439→    // their seats never come back within grab range
   440→    for (const s of this.swings) s.settle(dt, this.time)
   441→
   442→    // ---- jump with coyote time + input buffer (no double jumps) ----
   443→    this.coyote = b.onGround ? T.coyote : Math.max(0, this.coyote - dt)
   444→    if (jump) this.jumpBuf = T.buffer
   445→    else this.jumpBuf = Math.max(0, this.jumpBuf - dt)
   446→    if (this.jumpBuf > 0 && this.coyote > 0 && b.vy <= 0.01) {
   447→      b.vy = T.jumpVel
   448→      this.jumpBuf = 0
   449→      this.coyote = 0
   450→      b.onGround = false
   451→      this.duck.onJump(true)
   452→      this.sfx.jump()
   453→    }
   454→
   455→    // ---- gravity + collide ----
   456→    const fallBefore = b.vy
   457→    b.vy = Math.max(T.maxFall, b.vy + T.gravity * dt)
   458→    const was = b.onGround
   459→    const res = this.world.move(b, dt, was)
   460→    if (res.landed) {
   461→      const fs = Math.abs(Math.min(0, fallBefore))
   462→      if (fs > 2.5) this.sfx.land()
   463→      this.duck.onLand(fs)
   464→    }
   465→    if (res.landed && this.checkCheckpoints()) {
   466→      // a fresh checkpoint activation gets a tiny celebration
   467→      const cp = this.checkpoints[this.cpIndex]
   468→      this.duck.celebrate()
   469→      this.sfx.chime()
   470→      if (cp) this.burst(cp.x, cp.y + 1.3, 12)
   471→      this.cb.onTip('checkpoint')
   472→      this.cb.onCheckpoint(this.cpIndex + 1, this.checkpoints.length)
   473→    }
   474→
   475→    // ---- footsteps follow the run cycle ----
   476→    if (b.onGround && Math.abs(b.vx) > 0.8) {
   477→      if (this.duck.consumeStep()) {
   478→        this.stepAlt = !this.stepAlt
   479→        this.sfx.step(this.stepAlt)
   480→      }
   481→    }
   482→
   483→    // ---- swing interaction: nearest seat within ITS grab range, preferring
   484→    // the swing the duck is moving toward (never grabs a seat behind its
   485→    // back when the next swing ahead is also in range) ----
   486→    const moveDir = Math.abs(b.vx) > 0.5 ? Math.sign(b.vx) : 0
   487→    let nearest: Swing | null = null
   488→    let nearScore = Infinity
   489→    for (const s of this.swings) {
   490→      const seat = s.seatAnchor()
   491→      const d = Math.hypot(b.x - seat.x, b.y + 0.35 - seat.y)
   492→      const wasInRange = s.inRangeNow
   493→      const inRange = d < s.range && !(s === this.regrabSwing && this.regrabT > 0)
   494→      s.setInRange(inRange)
   495→      // first moment a swing becomes reachable: a soft glint + tiny sparkle
   496→      // helps players read WHERE the next interaction lives (edge-triggered,
   497→      // cooled down so cruising past swings never turns into a siren)
   498→      if (inRange && !wasInRange && !this.swinging) {
   499→        const last = this.glintCool.get(s) ?? -9
   500→        if (this.time - last > 1.4) {
   501→          this.glintCool.set(s, this.time)
   502→          this.sfx.glint()
   503→          this.burst(seat.x, seat.y + 0.3, 4)
   504→        }
   505→      }
   506→      if (inRange) {
   507→        const ahead = moveDir !== 0 ? Math.max(0, moveDir * (seat.x - b.x)) : 0
   508→        const score = d - 0.45 * Math.min(1.2, ahead)
   509→        if (score < nearScore) {
   510→          nearScore = score
   511→          nearest = s
   512→        }
   513→      }
   514→    }
   515→    if (nearest) {
   516→      this.setPrompt('GRAB')
   517→      if (interact) this.grab(nearest)
   518→    } else {
   519→      this.setPrompt('')
   520→    }
   521→
   522→    this.collectStars()
   523→
   524→    // ---- hazards: a thorn burr prick sends you back to the checkpoint.
   525→    // Checked every ground/air step (swing arcs stay above burr zones by
   526→    // design, so the attached branch skips this.)
   527→    if (this.checkHazards()) this.respawn(true)
   528→
   529→    // ---- safe recovery when falling out of the level ----
   530→    if (b.y < this.killY) this.respawn(true)
   531→
   532→    // ---- end flag ----
   533→    if (b.x > this.flagX - 0.35 && b.onGround) {
   534→      this.won = true
   535→      this.hopTimer = 0.3
   536→      this.sfx.win()
   537→      this.burst(b.x, b.y + 1.2, 26)
   538→      this.burst(b.x - 1.2, b.y + 2.2, 18)
   539→      this.burst(b.x + 1.2, b.y + 2.2, 18)
   540→      this.cb.onVictory(true, {
   541→        stars: this.starsGot,
   542→        starsTotal: this.level.stars.length,
   543→        levelIndex: this.levelIdx,
   544→        levelName: this.def.name,
   545→        time: this.levelTime,
   546→      })
   547→    }
   548→
   549→    this.syncDuck(dt, 0)
   550→    this.updateCamera(dt)
   551→  }
   552→
   553→  // ============================ hazards ============================
   554→
   555→  /** True when the duck touches a live hazard (its group tracks the patrol). */
   556→  private checkHazards(): boolean {
   557→    const hs = this.level?.hazards
   558→    if (!hs || hs.length === 0) return false
   559→    const b = this.body
   560→    for (const h of hs) {
   561→      const p = h.group.position
   562→      const dx = b.x - p.x
   563→      const dy = b.y - p.y
   564→      if (dx * dx + dy * dy < h.r * h.r) {
   565→        this.burst(p.x, p.y + 0.35, 8)
   566→        this.sfx.prick()
   567→        return true
   568→      }
   569→    }
   570→    return false
   571→  }

   573→  // ============================ stars ============================
   574→
   575→  private collectStars(): void {
   576→    const b = this.body
   577→    for (const s of this.level.stars) {
   578→      if (s.taken) continue
   579→      if (Math.hypot(b.x - s.x, b.y + 0.4 - s.y) < T.starRadius) {
   580→        s.taken = true
   581→        s.mesh.visible = false
   582→        this.starsGot++
   583→        this.sfx.collect()
   584→        this.burst(s.x, s.y, 10)
   585→        this.cb.onStars(this.starsGot, this.level.stars.length)
   586→        if (!this.tipSent.star) {
   587→          this.tipSent.star = true
   588→          this.cb.onTip('star')
   589→        }
   590→      }
   591→    }
   592→  }

   594→  // ============================ confetti (shared bursts) ============================
   595→
   596→  private burst(x: number, y: number, n: number): void {
   597→    for (let i = 0; i < n; i++) {
   598→      const m = new THREE.Mesh(
   599→        this.confettiGeo,
   600→        new THREE.MeshBasicMaterial({
   601→          color: this.confettiColors[i % this.confettiColors.length],
   602→          transparent: true,
   603→          opacity: 0.95,
   604→        }),
   605→      )
   606→      m.position.set(x + (Math.random() - 0.5) * 0.3, y + (Math.random() - 0.5) * 0.3, (Math.random() - 0.5) * 0.5)
   607→      m.scale.setScalar(0.7 + Math.random() * 0.9)
   608→      const a = Math.random() * Math.PI * 2
   609→      const sp = 1.6 + Math.random() * 2.6
   610→      this.confetti.push({
   611→        mesh: m,
   612→        vx: Math.cos(a) * sp,
   613→        vy: 2.4 + Math.random() * 3,
   614→        spin: (Math.random() - 0.5) * 16,
   615→        life: 0.75 + Math.random() * 0.4,
   616→        max: 1.15,
   617→        grow: 0.1,
   618→        gravity: 6.5,
   619→      })
   620→      this.scene.add(m)
   621→    }
   622→  }
   623→
   624→  private tickConfetti(dt: number): void {
   625→    for (let i = this.confetti.length - 1; i >= 0; i--) {
   626→      const p = this.confetti[i]
   627→      p.life -= dt
   628→      if (p.life <= 0) {
   629→        this.scene.remove(p.mesh)
   630→        ;(p.mesh.material as THREE.Material).dispose()
   631→        this.confetti.splice(i, 1)
   632→        continue
   633→      }
   634→      p.vy -= p.gravity * dt
   635→      p.vx *= 1 - 0.9 * dt
   636→      p.mesh.position.x += p.vx * dt
   637→      p.mesh.position.y += p.vy * dt
   638→      p.mesh.rotation.x += p.spin * dt
   639→      p.mesh.rotation.y += p.spin * 0.6 * dt
   640→      ;(p.mesh.material as THREE.MeshBasicMaterial).opacity = Math.min(1, p.life / 0.4)
   641→    }
   642→  }

   644→  // ============================ duck sync ============================
   645→
   646→  private syncDuck(dt: number, pump = 0): void {
   647→    const b = this.body
   648→    if (this.swinging && this.activeSwing) {
   649→      if (Math.abs(this.activeSwing.omega) > 0.25) this.duck.facing = this.activeSwing.omega > 0 ? 1 : -1
   650→    } else if (b.vx > 0.25) {
   651→      this.duck.facing = 1
   652→    } else if (b.vx < -0.25) {
   653→      this.duck.facing = -1
   654→    }
   655→    this.duck.root.position.set(b.x, b.y, 0)
   656→    // highest surface below the duck — feeds the contact shadow and
   657→    // gives the player a readable landing cue over gaps (movers included)
   658→    let groundY = -8
   659→    const consider = (x1: number, x2: number, top: number) => {
   660→      if (b.x >= x1 - 0.25 && b.x <= x2 + 0.25 && top <= b.y - b.hh + 0.08 && top > groundY) groundY = top
   661→    }
   662→    for (const p of this.plats) consider(p.x1, p.x2, p.top)
   663→    for (const box of this.moverBoxes) consider(box.cx - box.hw, box.cx + box.hw, box.cy + box.hh)
   664→    const speed = Math.hypot(b.vx, b.vy)
   665→    this.duck.update(dt, {
   666→      vx: b.vx,
   667→      vy: b.vy,
   668→      grounded: b.onGround,
   669→      swinging: this.swinging,
   670→      falling: !b.onGround && b.vy < -3,
   671→      fast: !b.onGround && !this.swinging && speed > 6.4,
   672→      tilt: this.swinging && this.activeSwing ? this.activeSwing.theta : 0,
   673→      pump,
   674→      won: this.won,
   675→      groundY,
   676→    })
   677→    this.dirLight.position.set(b.x + 5, 11, 8)
   678→    this.dirLight.target.position.set(b.x, 0, 0)
   679→  }
   680→
   681→  private updateCamera(dt: number): void {
   682→    const b = this.body
   683→    // look-ahead follows SMOOTHED velocity: anticipates launches and fast
   684→    // flights without ever snapping or oscillating
   685→    this.smVx += (b.vx - this.smVx) * Math.min(1, dt * 3.2)
   686→    const airborneFast = !b.onGround && !this.swinging && Math.abs(this.smVx) > 6
   687→    let lead = clamp(this.smVx * (airborneFast ? 0.62 : 0.45), -3.4, 3.4)
   688→    // while swinging, lean the frame into the arc's travel direction so the
   689→    // landing zone stays readable through the widest part of the sweep
   690→    if (this.swinging && this.activeSwing) {
   691→      lead += clamp(this.activeSwing.omega * this.activeSwing.L * 0.075, -1.1, 1.1)
   692→    }
   693→    const look = lead + this.duck.facing * 0.35
   694→    const tx = b.x + look
   695→    const ty = Math.max(0.5, b.y * 0.5 + 1.0) + (this.swinging ? 0.4 : 0)
   696→    const kx = 1 - Math.exp(-3.6 * dt)
   697→    const ky = 1 - Math.exp(-3 * dt)
   698→    this.camX += (tx - this.camX) * kx
   699→    this.camY += (ty - this.camY) * ky
   700→    // breathe out slightly while swinging to reveal the landing zone
   701→    const tz = this.camZ * (this.swinging ? 1.12 : 1)
   702→    this.camZCur += (tz - this.camZCur) * Math.min(1, dt * 2.5)
   703→    this.camera.position.set(this.camX, this.camY + 1.5, this.camZCur)
   704→    this.camera.lookAt(this.camX + 0.3, this.camY + 0.35, 0)
   705→  }

   707→  // ============================ swing chain ============================
   708→
   709→  private grab(sw: Swing): void {
   710→    const b = this.body
   711→    sw.theta = clamp(sw.angleFrom(b.x, b.y + 0.35), -1.2, 1.2)
   712→    sw.omega = clamp(b.vx / sw.L, -2.6, 2.6)
   713→    this.activeSwing = sw
   714→    this.swinging = true
   715→    b.vx = 0
   716→    b.vy = 0
   717→    b.onGround = false
   718→    for (const s of this.swings) s.setInRange(false)
   719→    sw.bump()
   720→    this.duck.onGrab()
   721→    this.sfx.grab()
   722→    this.sfx.sparkle()
   723→    if (!this.tipSent.pump) {
   724→      this.tipSent.pump = true
   725→      this.cb.onTip('pump')
   726→    }
   727→  }
   728→
   729→  private release(): void {
   730→    const sw = this.activeSwing
   731→    if (!sw) return
   732→    const v = sw.releaseVelocity()
   733→    const b = this.body
   734→    b.vx = v.x * T.releaseBoost
   735→    b.vy = Math.max(v.y * T.releaseBoost, -2)
   736→    // the duck kicks off the seat when releasing on a forward swing —
   737→    // a small, readable boost that rewards good timing
   738→    if (sw.omega > 0.3) b.vy += T.forwardKick
   739→    if (Math.abs(b.vx) < 2.4) {
   740→      b.vx = (sw.omega >= 0 ? 1 : -1) * 2.4
   741→    }
   742→    this.swinging = false
   743→    this.activeSwing = null
   744→    // brief cooldown so flying past THIS swing doesn't instantly re-grab it;
   745→    // any other swing in range stays grabbable for mid-air chains. The base
   746→    // cooldown scales with release speed: a hard launch leaves the old
   747→    // swing's grab zone later, so it must stay locked longer (a weak drop
   748→    // only locks it for a blink, keeping low-release re-grabs snappy).
   749→    this.regrabSwing = sw
   750→    this.regrabT = T.regrabCooldown + Math.min(0.55, Math.hypot(b.vx, b.vy) * 0.05)
   751→    b.onGround = false
   752→    this.duck.onJump()
   753→    this.duck.onRelease()
   754→    this.sfx.release()
   755→  }

   757→  // ============================ checkpoints / lifecycle ============================
   758→
   759→  /** Activate the next checkpoint when the duck lands on its platform. */
   760→  private checkCheckpoints(): boolean {
   761→    const b = this.body
   762→    for (let i = this.cpIndex + 1; i < this.checkpoints.length; i++) {
   763→      const cp = this.checkpoints[i]
   764→      const p = this.plats[cp.plat]
   765→      if (!p) continue
   766→      if (b.x > p.x1 + 0.3 && b.x < p.x2 - 0.3 && Math.abs(b.y - b.hh - cp.y) < 0.4) {
   767→        this.cpIndex = i
   768→        this.level?.setCheckpointActive(i)
   769→        return true
   770→      }
   771→    }
   772→    return false
   773→  }
   774→
   775→  respawn(withFlash: boolean): void {
   776→    const cp = this.cpIndex >= 0 ? this.checkpoints[this.cpIndex] : { x: this.spawnX, y: 0 }
   777→    const b = this.body
   778→    b.x = cp.x
   779→    b.y = cp.y + T.colliderH / 2 + 0.02
   780→    b.vx = 0
   781→    b.vy = 0
   782→    b.onGround = false
   783→    b.carrier = null
   784→    this.swinging = false
   785→    this.activeSwing = null
   786→    this.regrabSwing = null
   787→    this.regrabT = 0
   788→    this.debugRelease = null
   789→    for (const s of this.swings) s.setInRange(false)
   790→    this.duck.facing = 1
   791→    this.duck.dizzy()
   792→    this.camX = b.x
   793→    this.camY = Math.max(1.0, b.y)
   794→    this.sfx.fall()
   795→    if (withFlash) this.cb.onFlash()
   796→  }
   797→
   798→  reset(): void {
   799→    if (this.won) {
   800→      this.won = false
   801→      this.cb.onVictory(false)
   802→    }
   803→    this.hopTimer = 0
   804→    this.cpIndex = -1
   805→    this.starsGot = 0
   806→    this.levelTime = 0
   807→    this.tipSent = { pump: false, star: false }
   808→    for (const s of this.level.stars) {
   809→      s.taken = false
   810→      s.mesh.visible = true
   811→    }
   812→    this.cb.onStars(0, this.level.stars.length)
   813→    for (const s of this.swings) {
   814→      s.theta = 0
   815→      s.omega = 0
   816→    }
   817→    this.respawn(false)
   818→  }

   820→  /** Quick retry: back to the latest checkpoint, stars kept. */
   821→  retry(): void {
   822→    if (this.won) {
   823→      this.won = false
   824→      this.cb.onVictory(false)
   825→    }
   826→    this.respawn(true)
   827→  }

   867→  private key(e: KeyboardEvent, down: boolean): void {
   868→    if (e.repeat) return
   869→    const c = e.code
   870→    if (c === 'ArrowLeft' || c === 'ArrowRight' || c === 'ArrowUp' || c === 'Space') e.preventDefault()
   871→    if (down) this.sfx.unlock()
   872→    if (c === 'ArrowLeft' || c === 'KeyA') this.input.set('left', down)
   873→    else if (c === 'ArrowRight' || c === 'KeyD') this.input.set('right', down)
   874→    else if (c === 'Space' || c === 'ArrowUp' || c === 'KeyW') this.input.set('jump', down)
   875→    else if (c === 'KeyE' || c === 'Enter') this.input.set('interact', down)
   876→    else if (c === 'KeyR' && down) this.retry()
   877→  }
