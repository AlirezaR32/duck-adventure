    40→/** Smoothstep from a to b (works with b < a — the clamp flips it). */
    41→function smooth(a: number, b: number, t: number): number {
    42→  const k = Math.min(1, Math.max(0, (t - a) / (b - a)))
    43→  return k * k * (3 - 2 * k)
    44→}

    67→/**
    68→ * Smooth tube swept along a Catmull-Rom path with a radius that varies
    69→ * linearly across the `radii` keyframes (evenly spaced along the curve).
    70→ * Used for the neck so it can taper elegantly and flare into the head.
    71→ */
    72→function taperedTube(
    73→  pts: [number, number, number][],
    74→  radii: number[],
    75→  segs = 30,
    76→  radial = 22,
    77→): THREE.BufferGeometry {
    78→  const curve = new THREE.CatmullRomCurve3(pts.map((p) => new THREE.Vector3(p[0], p[1], p[2])))
    79→  const frames = curve.computeFrenetFrames(segs, false)
    80→  const pos: number[] = []
    81→  const idx: number[] = []
    82→  for (let i = 0; i <= segs; i++) {
    83→    const t = i / segs
    84→    const c = curve.getPoint(t)
    85→    const n = frames.normals[Math.min(i, segs - 1)]
    86→    const b = frames.binormals[Math.min(i, segs - 1)]
    87→    const f = t * (radii.length - 1)
    88→    const i0 = Math.floor(f)
    89→    const i1 = Math.min(radii.length - 1, i0 + 1)
    90→    const r = radii[i0] + (radii[i1] - radii[i0]) * (f - i0)
    91→    for (let j = 0; j <= radial; j++) {
    92→      const a = (j / radial) * Math.PI * 2
    93→      const ca = Math.cos(a)
    94→      const sa = Math.sin(a)
    95→      pos.push(
    96→        c.x + (n.x * ca + b.x * sa) * r,
    97→        c.y + (n.y * ca + b.y * sa) * r,
    98→        c.z + (n.z * ca + b.z * sa) * r,
    99→      )
   100→    }
   101→  }
   102→  const ring = radial + 1
   103→  for (let i = 0; i < segs; i++) {
   104→    for (let j = 0; j < radial; j++) {
   105→      const a = i * ring + j
   106→      idx.push(a, a + ring, a + 1, a + 1, a + ring, a + ring + 1)
   107→    }
   108→  }
   109→  const g = new THREE.BufferGeometry()
   110→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3))
   111→  g.setIndex(idx)
   112→  g.computeVertexNormals()
   113→  return g
   114→}

   116→/** Lathe from a (radius, height) profile. */
   117→function latheProfile(prof: [number, number][], segs = 36): THREE.LatheGeometry {
   118→  return new THREE.LatheGeometry(prof.map((p) => new THREE.Vector2(p[0], p[1])), segs)
   119→}

   121→/**
   122→ * Sculpt the raw pear lathe into the ornament silhouette:
   123→ *  - elongate fore-aft, slim side-to-side
   124→ *  - fill the low chest
   125→ *  - rotate the rear up around a mid-body pivot so the tail line is part
   126→ *    of the body itself (perky, tucked, smooth — no separate lump)
   127→ */
   128→function sculptTorso(geo: THREE.BufferGeometry): void {
   129→  const p = geo.getAttribute('position') as THREE.BufferAttribute
   130→  const v = new THREE.Vector3()
   131→  for (let i = 0; i < p.count; i++) {
   132→    v.fromBufferAttribute(p, i)
   133→    let x = v.x * 1.38
   134→    const y = v.y
   135→    const z = v.z * 0.92
   136→    x += smooth(0.05, 0.42, x) * smooth(0.5, 0.08, y) * 0.06 // chest fill
   137→    const rear = smooth(0.02, -0.55, x)
   138→    const a = -0.34 * rear // lift the rear
   139→    const py = y - 0.36
   140→    const ca = Math.cos(a)
   141→    const sa = Math.sin(a)
   142→    p.setXYZ(i, x * ca - py * sa, x * sa + py * ca + 0.36, z)
   143→  }
   144→  geo.computeVertexNormals()
   145→}

   147→/** Per-vertex tint along Y (lo at yLo → hi at yHi). Adds soft tonal depth. */
   148→function paintY(geo: THREE.BufferGeometry, yLo: number, yHi: number, lo = colLo, hi = colHi): void {
   149→  const pos = geo.getAttribute('position')
   150→  const col = new Float32Array(pos.count * 3)
   151→  const c = new THREE.Color()
   152→  for (let i = 0; i < pos.count; i++) {
   153→    const t = Math.min(1, Math.max(0, (pos.getY(i) - yLo) / (yHi - yLo)))
   154→    c.copy(lo).lerp(hi, t)
   155→    col[i * 3] = c.r
   156→    col[i * 3 + 1] = c.g
   157→    col[i * 3 + 2] = c.b
   158→  }
   159→  geo.setAttribute('color', new THREE.BufferAttribute(col, 3))
   160→}
   161→
   162→/** Per-vertex tint along X (used for the bill: warm base → deep tip). */
   163→function paintX(geo: THREE.BufferGeometry, xLo: number, xHi: number, lo: THREE.Color, hi: THREE.Color): void {
   164→  const pos = geo.getAttribute('position')
   165→  const col = new Float32Array(pos.count * 3)
   166→  const c = new THREE.Color()
   167→  for (let i = 0; i < pos.count; i++) {
   168→    const t = Math.min(1, Math.max(0, (pos.getX(i) - xLo) / (xHi - xLo)))
   169→    c.copy(lo).lerp(hi, t)
   170→    col[i * 3] = c.r
   171→    col[i * 3 + 1] = c.g
   172→    col[i * 3 + 2] = c.b
   173→  }
   174→  geo.setAttribute('color', new THREE.BufferAttribute(col, 3))
   175→}

   177→function blob(
   178→  r: number,
   179→  sx: number,
   180→  sy: number,
   181→  sz: number,
   182→  mat: THREE.Material,
   183→  segW = 24,
   184→  segH = 18,
   185→): THREE.Mesh {
   186→  const m = new THREE.Mesh(new THREE.SphereGeometry(r, segW, segH), mat)
   187→  m.scale.set(sx, sy, sz)
   188→  m.castShadow = true
   189→  return m
   190→}

   503→      bead.scale.set(1, 1.28, 0.62)
   504→      const glow = new THREE.Mesh(
   505→        new THREE.SphereGeometry(0.014, 8, 6),
   506→        new THREE.MeshBasicMaterial({ color: 0xffffff }),
   507→      )
   508→      glow.position.set(0.012, 0.017, zs > 0 ? 0.014 : -0.014)
   509→      g.add(bead, glow)
   510→      this.headGrp.add(g)
   511→      this.eyes.push(g)
   512→    }
   513→    this.neckGrp.add(this.headGrp)
   514→
   515→    // ---- stubby orange legs with webbed feet (on `squash` so landings
   516→    // compress the whole character; hips are the animation pivots) ----
   517→    const footShape = new THREE.Shape()
   518→    footShape.moveTo(0, 0.062)
   519→    footShape.quadraticCurveTo(0.14, 0.085, 0.235, 0.062)
   520→    footShape.quadraticCurveTo(0.16, 0.028, 0.185, 0.008)
   521→    footShape.quadraticCurveTo(0.245, -0.002, 0.245, -0.014)
   522→    footShape.quadraticCurveTo(0.16, -0.03, 0.175, -0.052)
   523→    footShape.quadraticCurveTo(0.2, -0.078, 0.185, -0.088)
   524→    footShape.quadraticCurveTo(0.08, -0.1, 0, -0.07)
   525→    footShape.quadraticCurveTo(-0.05, -0.005, 0, 0.062)
   526→    const footGeo = new THREE.ExtrudeGeometry(footShape, {
   527→      depth: 0.032,
   528→      bevelEnabled: true,
   529→      bevelThickness: 0.012,
   530→      bevelSize: 0.012,
   531→      bevelSegments: 2,
   532→      curveSegments: 8,
   533→    })
   534→    footGeo.rotateX(-Math.PI / 2)
   535→    for (const zs of [0.145, -0.145]) {
   536→      const hip = new THREE.Group()
   537→      hip.position.set(0.03, 0.28, zs)
   538→      const thigh = new THREE.Mesh(new THREE.CapsuleGeometry(0.055, 0.16, 6, 12), this.matFeet)
   539→      thigh.position.set(0, -0.1, 0)
   540→      thigh.castShadow = true
   541→      hip.add(thigh)
   542→      const ankle = new THREE.Group()
   543→      ankle.position.set(0.015, -0.27, 0)
   544→      const foot = new THREE.Mesh(footGeo, this.matFeet)
   545→      foot.position.y = 0.012
   546→      foot.castShadow = true
   547→      ankle.add(foot)
   548→      hip.add(ankle)
   549→      this.squash.add(hip)
   550→      this.hips.push(hip)
   551→      this.feet.push(ankle)
   552→    }
   553→  }

   666→  // ---------------- per-frame visual update ----------------
   667→
   668→  update(dt: number, s: DuckFrame): void {
   669→    this.t += dt
   670→
   671→    // contact shadow: hugs the ground below, fades as the duck climbs —
   672→    // doubles as a landing-spot readability cue over gaps
   673→    const h = Math.max(0, this.root.position.y - s.groundY)
   674→    const shadowVis = Math.max(0, 1 - h / 3.4)
   675→    const shadowMat = this.shadowMesh.material as THREE.MeshBasicMaterial
   676→    shadowMat.opacity = 0.9 * shadowVis
   677→    const shadowScale = 1 + Math.min(1.4, h * 0.16) * 0.3
   678→    this.shadowMesh.scale.set(shadowScale, shadowScale, 1)
   679→    this.shadowMesh.position.set(this.root.position.x, s.groundY + 0.035, 0)
   680→    this.shadowMesh.visible = shadowVis > 0.04 && s.groundY > -7
   681→
   682→    // squash & stretch spring (visual only — collider untouched)
   683→    const k = 240
   684→    const c = 17
   685→    this.sqVel += (-k * (this.sqY - 1) - c * this.sqVel) * dt
   686→    this.sqY += this.sqVel * dt
   687→    this.sqY = Math.min(1.32, Math.max(0.62, this.sqY))
   688→    const sx = 1 + (1 - this.sqY) * 0.62
   689→    this.squash.scale.set(sx, this.sqY, sx)
   690→
   691→    // facing ease (visual flip only)
   692→    const targetRot = this.facing === 1 ? 0 : Math.PI
   693→    this.root.rotation.y += (targetRot - this.root.rotation.y) * Math.min(1, dt * 14)
   694→
   695→    const speedAbs = Math.abs(s.vx)
   696→    const localVx = s.vx * this.facing
   697→    const runAmt = Math.min(1, speedAbs / 4.2)
   698→    this.celebT = Math.max(0, this.celebT - dt)
   699→    this.dizzyT = Math.max(0, this.dizzyT - dt)
   700→    this.flingT = Math.max(0, this.flingT - dt)
   701→    const celeb = this.celebT > 0 ? Math.sin((1 - this.celebT / 0.9) * Math.PI) : 0
   702→    const dizzy = this.dizzyT > 0 ? Math.sin(this.dizzyT / 1.15 * Math.PI) : 0
   703→    const fling = this.flingT > 0 ? Math.sin((1 - this.flingT / 0.5) * Math.PI) : 0
   704→
   705→    // pose blends
   706→    this.sitBlend += ((s.swinging ? 1 : 0) - this.sitBlend) * Math.min(1, dt * 9)
   707→    this.airBlend += ((s.grounded || s.swinging ? 0 : 1) - this.airBlend) * Math.min(1, dt * 9)
   708→
   709→    // run cycle phase
   710→    if (s.grounded && !s.swinging && speedAbs > 0.25) {
   711→      this.runPhase += dt * (5 + speedAbs * 2.4)
   712→    }
   713→    const w = this.runPhase
   714→
   715→    // body bounce + forward lean (+ celebration hop wiggle / dizzy wobble)
   716→    const bounce = s.grounded && !s.swinging ? Math.abs(Math.sin(w)) * 0.05 * runAmt : 0
   717→    const leanTarget = s.swinging
   718→      ? 0
   719→      : s.grounded
   720→        ? Math.max(-0.18, Math.min(0.18, localVx * 0.045))
   721→        : Math.max(-0.14, Math.min(0.14, localVx * 0.022)) + (s.falling ? 0.1 : -0.06)
   722→    this.lean += (leanTarget - this.lean) * Math.min(1, dt * 10)
   723→    this.bodyGrp.rotation.z =
   724→      -this.lean + 0.1 * this.sitBlend + Math.sin(this.t * 14) * 0.09 * celeb + Math.sin(this.t * 9) * 0.11 * dizzy
   725→    this.bodyGrp.position.y = bounce + celeb * 0.05
   726→    this.bodyGrp.rotation.y = Math.sin(this.t * 4.5) * 0.2 * dizzy
   727→
   728→    // gentle breathing while idle
   729→    const idle = s.grounded && speedAbs < 0.25 && !s.swinging
   730→    const breathe = idle ? Math.sin(this.t * 2.4) * 0.014 : 0
   731→    this.bodyGrp.scale.set(1 + breathe * 0.6, 1 + breathe, 1 + breathe * 0.6)
   732→
   733→    // neck: counter-sways while running, hangs along the rope while
   734→    // swinging (partial counter so the head leads the arc gracefully),
   735→    // curious tilts while idle, stretches forward in flight
   736→    let neckRot = -this.lean * 0.5
   737→    if (s.swinging) {
   738→      neckRot += -s.tilt * this.facing * 0.42 + s.pump * 0.14 + Math.sin(this.t * 1.8) * 0.03
   739→    }
   740→    if (this.airBlend > 0.5) neckRot += ((s.falling ? -0.06 : 0.16) + fling * 0.18) * this.airBlend
   741→    if (idle) {
   742→      this.curiosityIn -= dt
   743→      if (this.curiosityIn <= 0) {
   744→        this.curiosityTarget = (Math.random() - 0.5) * 0.34
   745→        this.curiosityIn = 2.4 + Math.random() * 3
   746→      }
   747→      this.curiosity += (this.curiosityTarget - this.curiosity) * Math.min(1, dt * 2.2)
   748→      neckRot += this.curiosity + Math.sin(this.t * 2.4) * 0.015
   749→    } else {
   750→      this.curiosity *= Math.max(0, 1 - dt * 5)
   751→    }
   752→    this.neckGrp.rotation.z += (neckRot - this.neckGrp.rotation.z) * Math.min(1, dt * 8)
   753→
   754→    // head: landing nod spring (+ celebration nod / dizzy head-shake),
   755→    // occasional idle glance toward the player
   756→    this.nodVel += (-160 * this.nod - 12 * this.nodVel) * dt
   757→    this.nod += this.nodVel * dt
   758→    this.headGrp.rotation.z =
   759→      Math.max(-0.5, Math.min(0.5, this.nod)) +
   760→      (s.won ? Math.sin(this.t * 6) * 0.06 : 0) +
   761→      celeb * Math.sin(this.t * 11) * 0.14 +
   762→      dizzy * Math.sin(this.t * 12) * 0.2
   763→    this.glanceIn -= dt
   764→    if (this.glanceIn <= 0 && idle) {
   765→      this.glanceTarget = (Math.random() > 0.5 ? 1 : -1) * (0.3 + Math.random() * 0.25)
   766→      this.glanceLeft = 0.9 + Math.random() * 0.4
   767→      this.glanceIn = 4 + Math.random() * 4
   768→    }
   769→    if (this.glanceLeft > 0) this.glanceLeft -= dt
   770→    const glance = idle && this.glanceLeft > 0 ? this.glanceTarget : 0
   771→    this.headGrp.rotation.y += (glance - this.headGrp.rotation.y) * Math.min(1, dt * 5)
   772→
   773→    // tail: happy wag on celebration/dizzy, a lazy sway when idle
   774→    const wagAmp = celeb * 0.5 + dizzy * 0.3
   775→    this.tailGrp.rotation.y = Math.sin(this.t * 11) * wagAmp + Math.sin(this.t * 1.3) * 0.06
   776→    this.tailGrp.rotation.z = 0.55 + celeb * 0.2
   777→
   778→    // rope alignment while swinging: the body hangs along the rope (world
   779→    // rotation = +theta; facing flips the local sign), eased softly
   780→    const tiltTarget = s.swinging ? s.tilt * this.facing * 0.78 : 0
   781→    this.squash.rotation.z += (tiltTarget - this.squash.rotation.z) * Math.min(1, dt * 12)
   782→    // sit deep onto the plank: butt back onto the seat, feet forward
   783→    this.squash.position.y += (-0.26 * this.sitBlend - this.squash.position.y) * Math.min(1, dt * 10)
   784→    this.squash.position.x += (-0.05 * this.sitBlend - this.squash.position.x) * Math.min(1, dt * 10)
   785→
   786→    // legs: walk cycle / airborne tuck & reach / forward sit dangle
   787→    const stepAmp = 0.55 * runAmt
   788→    const lag = s.swinging ? -s.tilt * this.facing * 0.3 + s.pump * 0.06 : 0
   789→    for (let i = 0; i < 2; i++) {
   790→      const hip = this.hips[i]
   791→      const ankle = this.feet[i]
   792→      const phase = w + (i === 0 ? 0 : Math.PI)
   793→      const standHip = Math.sin(phase) * stepAmp
   794→      const standFoot = -standHip * 0.85
   795→      const airHip = -0.38 + (s.falling ? 0.28 : 0) - fling * 0.3
   796→      const airFoot = s.falling ? 0.22 : 0.05
   797→      const sitHip = 1.15 + lag + Math.sin(this.t * 2.2 + i * 1.7) * 0.07
   798→      const sitFoot = -0.3 + Math.sin(this.t * 2.2 + i * 1.7 + 0.6) * 0.08
   799→      const targetHip = standHip + (airHip - standHip) * this.airBlend
   800→      const targetFoot = standFoot + (airFoot - standFoot) * this.airBlend
   801→      const blendedHip = targetHip + (sitHip - targetHip) * this.sitBlend
   802→      const blendedFoot = targetFoot + (sitFoot - targetFoot) * this.sitBlend
   803→      hip.rotation.z += (blendedHip - hip.rotation.z) * Math.min(1, dt * 14)
   804→      ankle.rotation.z += (blendedFoot - ankle.rotation.z) * Math.min(1, dt * 14)
   805→    }
   806→
   807→    // wings: tucked at rest · flutter on run · spread in air · hold the
   808→    // ropes when sitting · flare on release · big proud flaps when celebrating
   809→    let wingTarget: number
   810→    let wingSpeed = 0
   811→    if (s.won || celeb > 0.2) {
   812→      wingTarget = -1.5
   813→      wingSpeed = 15
   814→    } else if (fling > 0.15) {
   815→      wingTarget = -1.1
   816→      wingSpeed = 9
   817→    } else if (s.swinging) {
   818→      wingTarget = -2.45 + s.pump * 0.12
   819→    } else if (!s.grounded) {
   820→      wingTarget = s.fast ? -0.9 : -1.25
   821→      wingSpeed = s.fast ? 13 : 7
   822→    } else {
   823→      wingTarget = 0.05 + Math.abs(Math.sin(w)) * 0.1 * runAmt
   824→    }
   825→    for (let i = 0; i < 2; i++) {
   826→      const wing = this.wings[i]
   827→      const flap = wingSpeed > 0 ? Math.sin(this.t * wingSpeed + (i === 0 ? 0 : 0.35)) * 0.4 : 0
   828→      wing.rotation.z += (wingTarget + flap - wing.rotation.z) * Math.min(1, dt * 12)
   829→    }
   830→
   831→    // blink (wide eyes when falling, happy squint on victory)
   832→    this.blinkIn -= dt
   833→    if (this.blinkIn <= 0) {
   834→      this.blinkLeft = 0.12
   835→      this.blinkIn = 2 + Math.random() * 2.6
   836→    }
   837→    if (this.blinkLeft > 0) this.blinkLeft -= dt
   838→    const eyeY = s.won || celeb > 0.2 ? 0.42 : this.blinkLeft > 0 ? 0.12 : s.falling && !s.swinging ? 1.3 : 1
   839→    const eyeX = dizzy > 0 ? 1 + Math.sin(this.t * 16) * 0.3 : 1
   840→    for (const e of this.eyes) {
   841→      e.scale.y += (eyeY - e.scale.y) * Math.min(1, dt * 30)
   842→      e.scale.x += (eyeX - e.scale.x) * Math.min(1, dt * 20)
   843→    }
   844→
   845→    // wind streaks while flying fast
   846→    this.streakIn -= dt
   847→    if (s.fast && this.streakIn <= 0) {
   848→      this.streak()
   849→      this.streakIn = 0.055
   850→    }
   851→
   852→    // particles
   853→    for (let i = this.particles.length - 1; i >= 0; i--) {
   854→      const p = this.particles[i]
   855→      p.life -= dt
   856→      if (p.life <= 0) {
   857→        this.root.remove(p.mesh)
   858→        ;(p.mesh.material as THREE.Material).dispose()
   859→        this.particles.splice(i, 1)
   860→        continue
   861→      }
   862→      p.vy -= (p.grow < 1 ? 1.2 : 5) * dt
   863→      p.mesh.position.x += p.vx * dt
   864→      p.mesh.position.y += p.vy * dt
   865→      if (p.spin !== 0) {
   866→        p.mesh.rotation.x += p.spin * dt
   867→        p.mesh.rotation.y += p.spin * 0.7 * dt
   868→      }
   869→      const k2 = p.life / p.max
   870→      const mat = p.mesh.material as THREE.MeshBasicMaterial
   871→      mat.opacity = k2 * (p.grow < 1 ? 0.95 : 0.75)
   872→      p.mesh.scale.multiplyScalar(1 + p.grow * dt * (1 - k2))
   873→    }
   874→  }
