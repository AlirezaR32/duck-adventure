     1→/**
     2→ * Project Duck — game orchestrator (V0.4, chainable swings + progression).
     3→ *
     4→ * Fixed-timestep simulation (1/60) with an accumulator. The duck is a
     5→ * kinematic body: gravity, jump with anticipation, coyote time and jump
     6→ * buffering. Swinging is the star — and V0.4 makes it a CHAIN:
     7→ *
     8→ *  - every swing is independent (own pivot, length, damping, grab range)
     9→ *  - release preserves the pendulum's tangential momentum
    10→ *  - a short re-grab cooldown on the swing you just left prevents accidental
    11→ *    re-attaches, while any OTHER swing in range can be grabbed instantly
    12→ *  - explicit checkpoint posts activate as you pass them; respawning resets
    13→ *    physics, swing attachments and input safely (never mid-air stale state)
    14→ *
    15→ * Progression is lightweight: optional stars along the arcs, a completion
    16→ * screen with the star count, and best-progress persisted by the UI layer.
    17→ */
    18→

    28→const T = {
    29→  gravity: -30,
    30→  maxFall: -24,
    31→  moveSpeed: 4.2,
    32→  groundAccel: 34,
    33→  groundDecel: 38,
    34→  airAccel: 15,
    35→  airDrag: 0.6, // light: preserve swing-release momentum through the air
    36→  jumpVel: 11.2,
    37→  coyote: 0.09,
    38→  buffer: 0.12,
    39→  colliderW: 0.6,
    40→  colliderH: 0.78,
    41→  swingPump: 4.6,
    42→  releaseBoost: 1.12,
    43→  forwardKick: 1.15,
    44→  regrabCooldown: 0.24,
    45→  starRadius: 0.95,
    46→  fixedDt: 1 / 60,
    47→}
    48→
    49→function moveToward(v: number, target: number, maxDelta: number): number {
    50→  return v < target ? Math.min(v + maxDelta, target) : Math.max(v - maxDelta, target)
    51→}
    52→
    53→function clamp(v: number, lo: number, hi: number): number {
    54→  return Math.max(lo, Math.min(hi, v))
    55→}

   217→  // ============================ main loop ============================
   218→
   219→  private tick = (now: number): void => {
   220→    if (this.disposed) return
   221→    this.raf = requestAnimationFrame(this.tick)
   222→    let dt = (now - this.lastT) / 1000
   223→    this.lastT = now
   224→    dt = Math.min(dt, 0.1)
   225→    this.acc += dt
   226→    let steps = 0
   227→    while (this.acc >= T.fixedDt && steps < 5) {
   228→      this.step(T.fixedDt)
   229→      this.acc -= T.fixedDt
   230→      steps++
   231→    }
   232→    if (steps === 5) this.acc = 0
   233→    this.bg.update(dt, this.camX)
   234→    this.level.update(this.time)
   235→    for (const s of this.swings) s.tickVisual(dt, this.time)
   236→    this.tickConfetti(dt)
   237→    this.renderer.render(this.scene, this.camera)
   238→  }
   239→
   240→  private step(dt: number): void {
   241→    this.time += dt
   242→    const b = this.body
   243→    const interact = this.input.consumeInteract()
   244→    const jump = this.input.consumeJump()
   245→    this.regrabT = Math.max(0, this.regrabT - dt)
   246→    if (this.regrabT === 0) this.regrabSwing = null
   247→
   248→    if (this.won) {
   249→      // celebratory hops at the flag
   250→      this.hopTimer -= dt
   251→      if (b.onGround && this.hopTimer <= 0) {
   252→        b.vy = 7.2
   253→        b.onGround = false
   254→        this.duck.onJump()
   255→        this.hopTimer = 0.55
   256→      }
   257→      b.vx = moveToward(b.vx, 0, T.groundDecel * dt)
   258→      const was = b.onGround
   259→      b.vy = Math.max(T.maxFall, b.vy + T.gravity * dt)
   260→      this.world.move(b, dt, was)
   261→      for (const s of this.swings) s.settle(dt, this.time)
   262→      this.syncDuck(dt)
   263→      this.updateCamera(dt)
   264→      return
   265→    }
   266→
   267→    if (this.swinging && this.activeSwing) {
   268→      const sw = this.activeSwing
   269→      // pumping: pushing in the direction of motion adds energy (leg pumping)
   270→      const dir = this.input.dir()
   271→      const align = dir !== 0 ? (Math.sign(sw.omega) === dir ? 1 : 0.25) : 0
   272→      const pump = dir * align * T.swingPump
   273→      sw.update(dt, pump, 30)
   274→      for (const s of this.swings) if (s !== sw) s.settle(dt, this.time)
   275→      if (this.debugRelease) {
   276→        const d = this.debugRelease
   277→        if (sw.omega > d.minOmega && sw.theta > d.lo && sw.theta < d.hi) {
   278→          this.debugRelease = null
   279→          this.release()
   280→        }
   281→      }
   282→      const a = sw.seatAnchor()
   283→      b.x = a.x
   284→      b.y = a.y + 0.08 // feet rest on the seat board
   285→      // track tangential velocity so release preserves momentum
   286→      b.vx = sw.omega * sw.L * Math.cos(sw.theta)
   287→      b.vy = sw.omega * sw.L * Math.sin(sw.theta)
   288→      this.setPrompt('LET GO')
   289→      if (jump || interact) this.release()
   290→      // airy whoosh when sweeping through the bottom of the arc at speed
   291→      this.whooshT -= dt
   292→      if (Math.abs(sw.omega) > 3.1 && Math.abs(sw.theta) < 0.24 && this.whooshT <= 0) {
   293→        this.sfx.whoosh()
   294→        this.whooshT = 0.6
   295→      }
   296→      this.collectStars()
   297→      this.syncDuck(dt, pump / T.swingPump)
   298→      this.updateCamera(dt)
   299→      return
   300→    }
   301→
   302→    // ---- horizontal movement ----
   303→    const dir = this.input.dir()
   304→    if (dir !== 0) {
   305→      const target = dir * T.moveSpeed
   306→      const sameDir = Math.sign(b.vx) === dir
   307→      const over = Math.abs(b.vx) > T.moveSpeed + 0.05
   308→      if (over && sameDir) {
   309→        // preserve hard-earned momentum (e.g. swing releases); settle
   310→        // gently back to run speed only once grounded
   311→        if (b.onGround) b.vx = moveToward(b.vx, target, 8 * dt)
   312→      } else {
   313→        const rate = b.onGround ? T.groundAccel : T.airAccel
   314→        b.vx = moveToward(b.vx, target, rate * dt)
   315→      }
   316→    } else {
   317→      b.vx = moveToward(b.vx, 0, (b.onGround ? T.groundDecel : T.airDrag) * dt)
   318→    }
   319→
   320→    // unattended swings keep settling/swaying between grabs — released
   321→    // swings must actually lose energy, otherwise they spin forever and
   322→    // their seats never come back within grab range
   323→    for (const s of this.swings) s.settle(dt, this.time)
   324→
   325→    // ---- jump with coyote time + input buffer (no double jumps) ----
   326→    this.coyote = b.onGround ? T.coyote : Math.max(0, this.coyote - dt)
   327→    if (jump) this.jumpBuf = T.buffer
   328→    else this.jumpBuf = Math.max(0, this.jumpBuf - dt)
   329→    if (this.jumpBuf > 0 && this.coyote > 0 && b.vy <= 0.01) {
   330→      b.vy = T.jumpVel
   331→      this.jumpBuf = 0
   332→      this.coyote = 0
   333→      b.onGround = false
   334→      this.duck.onJump()
   335→      this.sfx.jump()
   336→    }
   337→
   338→    // ---- gravity + collide ----
   339→    const fallBefore = b.vy
   340→    b.vy = Math.max(T.maxFall, b.vy + T.gravity * dt)
   341→    const was = b.onGround
   342→    const res = this.world.move(b, dt, was)
   343→    if (res.landed) {
   344→      const fs = Math.abs(Math.min(0, fallBefore))
   345→      if (fs > 2.5) this.sfx.land()
   346→      this.duck.onLand(fs)
   347→    }
   348→    if (res.landed && this.checkCheckpoints()) {
   349→      // a fresh checkpoint activation gets a tiny celebration
   350→      this.duck.celebrate()
   351→      this.sfx.chime()
   352→      this.cb.onTip('checkpoint')
   353→      this.cb.onCheckpoint(this.cpIndex + 1, CHECKPOINTS.length)
   354→    }
   355→
   356→    // ---- footsteps follow the run cycle ----
   357→    if (b.onGround && Math.abs(b.vx) > 0.8) {
   358→      if (this.duck.consumeStep()) {
   359→        this.stepAlt = !this.stepAlt
   360→        this.sfx.step(this.stepAlt)
   361→      }
   362→    }
   363→
   364→    // ---- swing interaction: nearest seat within ITS grab range, preferring
   365→    // the swing the duck is moving toward (never grabs a seat behind its
   366→    // back when the next swing ahead is also in range) ----
   367→    const moveDir = Math.abs(b.vx) > 0.5 ? Math.sign(b.vx) : 0
   368→    let nearest: Swing | null = null
   369→    let nearScore = Infinity
   370→    for (const s of this.swings) {
   371→      const seat = s.seatAnchor()
   372→      const d = Math.hypot(b.x - seat.x, b.y + 0.35 - seat.y)
   373→      const inRange = d < s.range && !(s === this.regrabSwing && this.regrabT > 0)
   374→      s.setInRange(inRange)
   375→      if (inRange) {
   376→        const ahead = moveDir !== 0 ? Math.max(0, moveDir * (seat.x - b.x)) : 0
   377→        const score = d - 0.45 * Math.min(1.2, ahead)
   378→        if (score < nearScore) {
   379→          nearScore = score
   380→          nearest = s
   381→        }
   382→      }
   383→    }
   384→    if (nearest) {
   385→      this.setPrompt('GRAB')
   386→      if (interact) this.grab(nearest)
   387→    } else {
   388→      this.setPrompt('')
   389→    }
   390→
   391→    this.collectStars()
   392→
   393→    // ---- safe recovery when falling out of the level ----
   394→    if (b.y < KILL_Y) this.respawn(true)
   395→
   396→    // ---- end flag ----
   397→    if (b.x > FLAG_X - 0.35 && b.onGround) {
   398→      this.won = true
   399→      this.hopTimer = 0.3
   400→      this.sfx.win()
   401→      this.burst(b.x, b.y + 1.2, 26)
   402→      this.burst(b.x - 1.2, b.y + 2.2, 18)
   403→      this.burst(b.x + 1.2, b.y + 2.2, 18)
   404→      this.cb.onVictory(true, { stars: this.starsGot, starsTotal: this.level.stars.length })
   405→    }
   406→
   407→    this.syncDuck(dt, 0)
   408→    this.updateCamera(dt)
   409→  }
   410→
   411→  // ============================ stars ============================
   412→
   413→  private collectStars(): void {
   414→    const b = this.body
   415→    for (const s of this.level.stars) {
   416→      if (s.taken) continue
   417→      if (Math.hypot(b.x - s.x, b.y + 0.4 - s.y) < T.starRadius) {
   418→        s.taken = true
   419→        s.mesh.visible = false
   420→        this.starsGot++
   421→        this.sfx.collect()
   422→        this.burst(s.x, s.y, 10)
   423→        this.cb.onStars(this.starsGot, this.level.stars.length)
   424→        if (!this.tipSent.star) {
   425→          this.tipSent.star = true
   426→          this.cb.onTip('star')
   427→        }
   428→      }
   429→    }
   430→  }

   432→  // ============================ confetti (shared bursts) ============================
   433→
   434→  private burst(x: number, y: number, n: number): void {
   435→    for (let i = 0; i < n; i++) {
   436→      const m = new THREE.Mesh(
   437→        this.confettiGeo,
   438→        new THREE.MeshBasicMaterial({
   439→          color: this.confettiColors[i % this.confettiColors.length],
   440→          transparent: true,
   441→          opacity: 0.95,
   442→        }),
   443→      )
   444→      m.position.set(x + (Math.random() - 0.5) * 0.3, y + (Math.random() - 0.5) * 0.3, (Math.random() - 0.5) * 0.5)
   445→      m.scale.setScalar(0.7 + Math.random() * 0.9)
   446→      const a = Math.random() * Math.PI * 2
   447→      const sp = 1.6 + Math.random() * 2.6
   448→      this.confetti.push({
   449→        mesh: m,
   450→        vx: Math.cos(a) * sp,
   451→        vy: 2.4 + Math.random() * 3,
   452→        spin: (Math.random() - 0.5) * 16,
   453→        life: 0.75 + Math.random() * 0.4,
   454→        max: 1.15,
   455→        grow: 0.1,
   456→        gravity: 6.5,
   457→      })
   458→      this.scene.add(m)
   459→    }
   460→  }
   461→
   462→  private tickConfetti(dt: number): void {
   463→    for (let i = this.confetti.length - 1; i >= 0; i--) {
   464→      const p = this.confetti[i]
   465→      p.life -= dt
   466→      if (p.life <= 0) {
   467→        this.scene.remove(p.mesh)
   468→        ;(p.mesh.material as THREE.Material).dispose()
   469→        this.confetti.splice(i, 1)
   470→        continue
   471→      }
   472→      p.vy -= p.gravity * dt
   473→      p.vx *= 1 - 0.9 * dt
   474→      p.mesh.position.x += p.vx * dt
   475→      p.mesh.position.y += p.vy * dt
   476→      p.mesh.rotation.x += p.spin * dt
   477→      p.mesh.rotation.y += p.spin * 0.6 * dt
   478→      ;(p.mesh.material as THREE.MeshBasicMaterial).opacity = Math.min(1, p.life / 0.4)
   479→    }
   480→  }

   482→  // ============================ duck sync ============================
   483→
   484→  private syncDuck(dt: number, pump = 0): void {
   485→    const b = this.body
   486→    if (this.swinging && this.activeSwing) {
   487→      if (Math.abs(this.activeSwing.omega) > 0.25) this.duck.facing = this.activeSwing.omega > 0 ? 1 : -1
   488→    } else if (b.vx > 0.25) {
   489→      this.duck.facing = 1
   490→    } else if (b.vx < -0.25) {
   491→      this.duck.facing = -1
   492→    }
   493→    this.duck.root.position.set(b.x, b.y, 0)
   494→    // highest platform top below the duck — feeds the contact shadow and
   495→    // gives the player a readable landing cue over gaps
   496→    let groundY = -8
   497→    for (const p of PLATFORMS) {
   498→      if (b.x >= p.x1 - 0.25 && b.x <= p.x2 + 0.25 && p.top <= b.y - b.hh + 0.08 && p.top > groundY) {
   499→        groundY = p.top
   500→      }
   501→    }
   502→    const speed = Math.hypot(b.vx, b.vy)
   503→    this.duck.update(dt, {
   504→      vx: b.vx,
   505→      grounded: b.onGround,
   506→      swinging: this.swinging,
   507→      falling: !b.onGround && b.vy < -3,
   508→      fast: !b.onGround && !this.swinging && speed > 6.4,
   509→      tilt: this.swinging && this.activeSwing ? this.activeSwing.theta : 0,
   510→      pump,
   511→      won: this.won,
   512→      groundY,
   513→    })
   514→    this.dirLight.position.set(b.x + 5, 11, 8)
   515→    this.dirLight.target.position.set(b.x, 0, 0)
   516→  }
   517→
   518→  private updateCamera(dt: number): void {
   519→    const b = this.body
   520→    // look-ahead follows SMOOTHED velocity: anticipates launches and fast
   521→    // flights without ever snapping or oscillating
   522→    this.smVx += (b.vx - this.smVx) * Math.min(1, dt * 3.2)
   523→    const airborneFast = !b.onGround && !this.swinging && Math.abs(this.smVx) > 6
   524→    const lead = clamp(this.smVx * (airborneFast ? 0.62 : 0.45), -3.4, 3.4)
   525→    const look = lead + this.duck.facing * 0.35
   526→    const tx = b.x + look
   527→    const ty = Math.max(0.5, b.y * 0.5 + 1.0) + (this.swinging ? 0.4 : 0)
   528→    const kx = 1 - Math.exp(-3.6 * dt)
   529→    const ky = 1 - Math.exp(-3 * dt)
   530→    this.camX += (tx - this.camX) * kx
   531→    this.camY += (ty - this.camY) * ky
   532→    // breathe out slightly while swinging to reveal the landing zone
   533→    const tz = this.camZ * (this.swinging ? 1.12 : 1)
   534→    this.camZCur += (tz - this.camZCur) * Math.min(1, dt * 2.5)
   535→    this.camera.position.set(this.camX, this.camY + 1.5, this.camZCur)
   536→    this.camera.lookAt(this.camX + 0.3, this.camY + 0.35, 0)
   537→  }
   538→
   539→  // ============================ swing chain ============================
   540→
   541→  private grab(sw: Swing): void {
   542→    const b = this.body
   543→    sw.theta = clamp(sw.angleFrom(b.x, b.y + 0.35), -1.2, 1.2)
   544→    sw.omega = clamp(b.vx / sw.L, -2.6, 2.6)
   545→    this.activeSwing = sw
   546→    this.swinging = true
   547→    b.vx = 0
   548→    b.vy = 0
   549→    b.onGround = false
   550→    for (const s of this.swings) s.setInRange(false)
   551→    sw.bump()
   552→    this.duck.onGrab()
   553→    this.sfx.grab()
   554→    this.sfx.sparkle()
   555→    if (!this.tipSent.pump) {
   556→      this.tipSent.pump = true
   557→      this.cb.onTip('pump')
   558→    }
   559→  }
   560→
   561→  private release(): void {
   562→    const sw = this.activeSwing
   563→    if (!sw) return
   564→    const v = sw.releaseVelocity()
   565→    const b = this.body
   566→    b.vx = v.x * T.releaseBoost
   567→    b.vy = Math.max(v.y * T.releaseBoost, -2)
   568→    // the duck kicks off the seat when releasing on a forward swing —
   569→    // a small, readable boost that rewards good timing
   570→    if (sw.omega > 0.3) b.vy += T.forwardKick
   571→    if (Math.abs(b.vx) < 2.4) {
   572→      b.vx = (sw.omega >= 0 ? 1 : -1) * 2.4
   573→    }
   574→    this.swinging = false
   575→    this.activeSwing = null
   576→    // brief cooldown so flying past THIS swing doesn't instantly re-grab it;
   577→    // any other swing in range stays grabbable for mid-air chains
   578→    this.regrabSwing = sw
   579→    this.regrabT = T.regrabCooldown
   580→    b.onGround = false
   581→    this.duck.onJump()
   582→    this.duck.onRelease()
   583→    this.sfx.release()
   584→  }

   586→  // ============================ checkpoints / lifecycle ============================
   587→
   588→  /** Activate the next checkpoint when the duck lands on its platform. */
   589→  private checkCheckpoints(): boolean {
   590→    const b = this.body
   591→    for (let i = this.cpIndex + 1; i < CHECKPOINTS.length; i++) {
   592→      const cp = CHECKPOINTS[i]
   593→      const p = PLATFORMS[cp.plat]
   594→      if (!p) continue
   595→      if (b.x > p.x1 + 0.3 && b.x < p.x2 - 0.3 && Math.abs(b.y - b.hh - cp.y) < 0.4) {
   596→        this.cpIndex = i
   597→        this.level.setCheckpointActive(i)
   598→        return true
   599→      }
   600→    }
   601→    return false
   602→  }
   603→
   604→  respawn(withFlash: boolean): void {
   605→    const cp = this.cpIndex >= 0 ? CHECKPOINTS[this.cpIndex] : { x: SPAWN.x, y: 0 }
   606→    const b = this.body
   607→    b.x = cp.x
   608→    b.y = cp.y + T.colliderH / 2 + 0.02
   609→    b.vx = 0
   610→    b.vy = 0
   611→    b.onGround = false
   612→    this.swinging = false
   613→    this.activeSwing = null
   614→    this.regrabSwing = null
   615→    this.regrabT = 0
   616→    this.debugRelease = null
   617→    for (const s of this.swings) s.setInRange(false)
   618→    this.duck.facing = 1
   619→    this.duck.dizzy()
   620→    this.camX = b.x
   621→    this.camY = Math.max(1.0, b.y)
   622→    this.sfx.fall()
   623→    if (withFlash) this.cb.onFlash()
   624→  }
   625→
   626→  reset(): void {
   627→    if (this.won) {
   628→      this.won = false
   629→      this.cb.onVictory(false)
   630→    }
   631→    this.hopTimer = 0
   632→    this.cpIndex = -1
   633→    this.starsGot = 0
   634→    this.tipSent = { pump: false, star: false }
   635→    for (const s of this.level.stars) {
   636→      s.taken = false
   637→      s.mesh.visible = true
   638→    }
   639→    this.cb.onStars(0, this.level.stars.length)
   640→    for (const s of this.swings) {
   641→      s.theta = 0
   642→      s.omega = 0
   643→    }
   644→    this.respawn(false)
   645→  }
   646→
   647→  /** Quick retry: back to the latest checkpoint, stars kept. */
   648→  retry(): void {
   649→    if (this.won) {
   650→      this.won = false
   651→      this.cb.onVictory(false)
   652→    }
   653→    this.respawn(true)
   654→  }
   655→
   656→  /** Full restart from the very beginning (PLAY AGAIN). */
   657→  fullReset(): void {
   658→    this.reset()
   659→  }

   663→      this.promptLabel = label
   664→      this.cb.onPrompt(label !== '', label)

   668→  resize(): void {
   669→    const w = window.innerWidth
   670→    const h = window.innerHeight
   671→    this.renderer.setSize(w, h, false)
   672→    this.camera.aspect = w / h
   673→    const t = Math.tan(THREE.MathUtils.degToRad(21))
   674→    const aspect = w / h
   675→    // fit: at least ~8.8 units wide (~7.2 on tall portrait phones) and
   676→    // ~5.6 units tall of gameplay view
   677→    const fitW = aspect < 0.75 ? 7.2 : 8.8
   678→    this.camZ = Math.min(30, Math.max(fitW / (2 * t * aspect), 5.6 / (2 * t)))
   679→    this.camera.updateProjectionMatrix()
   680→  }

   694→  private key(e: KeyboardEvent, down: boolean): void {
   695→    if (e.repeat) return
   696→    const c = e.code
   697→    if (c === 'ArrowLeft' || c === 'ArrowRight' || c === 'ArrowUp' || c === 'Space') e.preventDefault()
   698→    if (down) this.sfx.unlock()
   699→    if (c === 'ArrowLeft' || c === 'KeyA') this.input.set('left', down)
   700→    else if (c === 'ArrowRight' || c === 'KeyD') this.input.set('right', down)
   701→    else if (c === 'Space' || c === 'ArrowUp' || c === 'KeyW') this.input.set('jump', down)
   702→    else if (c === 'KeyE' || c === 'Enter') this.input.set('interact', down)
   703→    else if (c === 'KeyR' && down) this.retry()
   704→  }

   790→          checkpoints: CHECKPOINTS.length,
