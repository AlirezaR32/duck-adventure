     1→/**
     2→ * Project Duck — game orchestrator (V0.6, multi-level).
     3→ *
     4→ * Fixed-timestep simulation (1/60) with an accumulator. The duck is a
     5→ * kinematic body: gravity, jump with anticipation, coyote time and jump
     6→ * buffering. Swinging is the star — grab, pump, release, fly, land:
     7→ *
     8→ *  - every swing is independent (own pivot, length, damping, grab range)
     9→ *  - release preserves the pendulum's tangential momentum
    10→ *  - a short re-grab cooldown on the swing you just left prevents accidental
    11→ *    re-attaches, while any OTHER swing in range can be grabbed instantly
    12→ *  - explicit checkpoint posts activate as you pass them; respawning resets
    13→ *    physics, swing attachments and input safely (never mid-air stale state)
    14→ *  - moving platforms carry the duck via the physics carrier (deterministic)
    15→ *
    16→ * V0.6 adds a reusable level system: levels are data (levels.ts) built by
    17→ * buildLevelRuntime; loadLevel() tears the previous level down completely
    18→ * (scene objects, physics boxes, swings, progression) so repeated switches
    19→ * and restarts can never leak stale state. One Game instance serves the
    20→ * whole session — input handlers and the RAF loop are never duplicated.
    21→ */
    22→

    35→  maxFall: -24,
    36→  moveSpeed: 4.2,
    37→  groundAccel: 34,
    38→  groundDecel: 38,
    39→  airAccel: 15,

    41→  jumpVel: 11.2,

    53→
    54→function moveToward(v: number, target: number, maxDelta: number): number {
    55→  return v < target ? Math.min(v + maxDelta, target) : Math.max(v - maxDelta, target)
    56→}
    57→
    58→function clamp(v: number, lo: number, hi: number): number {
    59→  return Math.max(lo, Math.min(hi, v))
    60→}

   244→  /** Tear the current level down completely, then build level `idx` fresh. */
   245→  loadLevel(idx: number): void {
   246→    const i = Math.max(0, Math.min(LEVELS.length - 1, Math.floor(idx)))
   247→    this.levelIdx = i
   248→    const def = LEVELS[i]
   249→    this.def = def
   250→
   251→    // ---- teardown: scene objects, swings, physics ----
   252→    if (this.level) this.level.dispose()
   253→    for (const s of this.swings) {
   254→      this.scene.remove(s.group)
   255→      s.dispose()
   256→    }
   257→    this.swings = []
   258→    this.world.reset()
   259→
   260→    // ---- build ----
   261→    this.level = buildLevelRuntime(this.scene, this.world, def)
   262→    for (const cfg of def.swings) {
   263→      const s = new Swing(cfg)
   264→      this.swings.push(s)
   265→      this.scene.add(s.group)
   266→    }
   267→    this.moverBoxes = this.world.boxes.filter((b) => this.world.motionOf(b) !== undefined)
   268→
   269→    // ---- per-level play state (no stale forces, attachments or counters) ----
   270→    this.plats = def.platforms
   271→    this.checkpoints = def.checkpoints
   272→    this.flagX = def.flagX
   273→    this.spawnX = def.spawnX
   274→    this.killY = def.killY
   275→    this.levelTime = 0
   276→    this.cpIndex = -1
   277→    this.starsGot = 0
   278→    this.tipSent = { pump: false, star: false }
   279→    this.clearConfetti()
   280→    this.won = false
   281→    this.hopTimer = 0
   282→    this.swinging = false
   283→    this.activeSwing = null
   284→    this.regrabSwing = null
   285→    this.regrabT = 0
   286→    this.whooshT = 0
   287→    this.debugRelease = null
   288→    this.input.clearAll()

   309→    for (const p of this.confetti) {

   316→  // ============================ main loop ============================
   317→
   318→  private tick = (now: number): void => {
   319→    if (this.disposed) return
   320→    this.raf = requestAnimationFrame(this.tick)
   321→    let dt = (now - this.lastT) / 1000
   322→    this.lastT = now
   323→    dt = Math.min(dt, 0.1)
   324→    this.acc += dt
   325→    let steps = 0
   326→    while (this.acc >= T.fixedDt && steps < 5) {
   327→      this.step(T.fixedDt)
   328→      this.acc -= T.fixedDt
   329→      steps++
   330→    }
   331→    if (steps === 5) this.acc = 0
   332→    this.bg.update(dt, this.camX)
   333→    this.level.update(this.time)
   334→    for (const s of this.swings) s.tickVisual(dt, this.time)
   335→    this.tickConfetti(dt)
   336→    this.renderer.render(this.scene, this.camera)
   337→  }
   338→
   339→  private step(dt: number): void {
   340→    this.time += dt
   341→    // moving platforms advance once per fixed step, before anything moves;
   342→    // a grounded duck then rides its carrier by the platform's exact delta
   343→    this.world.updateMotions(this.time)
   344→    const b = this.body
   345→    if (b.onGround && b.carrier) {
   346→      const m = this.world.motionOf(b.carrier)
   347→      if (m) {
   348→        b.x += m.dx
   349→        b.y += m.dy
   350→      }
   351→    }
   352→    if (!this.won) this.levelTime += dt
   353→    const interact = this.input.consumeInteract()
   354→    const jump = this.input.consumeJump()
   355→    this.regrabT = Math.max(0, this.regrabT - dt)
   356→    if (this.regrabT === 0) this.regrabSwing = null
   357→
   358→    if (this.won) {
   359→      // celebratory hops at the flag
   360→      this.hopTimer -= dt
   361→      if (b.onGround && this.hopTimer <= 0) {
   362→        b.vy = 7.2
   363→        b.onGround = false
   364→        this.duck.onJump()
   365→        this.hopTimer = 0.55
   366→      }
   367→      b.vx = moveToward(b.vx, 0, T.groundDecel * dt)
   368→      const was = b.onGround
   369→      b.vy = Math.max(T.maxFall, b.vy + T.gravity * dt)
   370→      this.world.move(b, dt, was)
   371→      for (const s of this.swings) s.settle(dt, this.time)
   372→      this.syncDuck(dt)
   373→      this.updateCamera(dt)
   374→      return
   375→    }
   376→
   377→    if (this.swinging && this.activeSwing) {
   378→      const sw = this.activeSwing
   379→      // pumping: pushing in the direction of motion adds energy (leg pumping)
   380→      const dir = this.input.dir()
   381→      const align = dir !== 0 ? (Math.sign(sw.omega) === dir ? 1 : 0.25) : 0
   382→      const pump = dir * align * T.swingPump
   383→      sw.update(dt, pump, 30)
   384→      for (const s of this.swings) if (s !== sw) s.settle(dt, this.time)
   385→      // the ropes creak softly when the duck works the swing hard or sweeps
   386→      // near the top of the arc — quiet, occasional, part of the world
   387→      this.creakT -= dt
   388→      if ((Math.abs(pump) > 2.2 || Math.abs(sw.theta) > 1.02) && this.creakT <= 0) {
   389→        this.sfx.creak()
   390→        this.creakT = 0.5 + Math.random() * 0.4
   391→      }
   392→      if (this.debugRelease) {
   393→        const d = this.debugRelease
   394→        if (sw.omega > d.minOmega && sw.theta > d.lo && sw.theta < d.hi) {
   395→          this.debugRelease = null
   396→          this.release()
   397→        }
   398→      }
   399→      const a = sw.seatAnchor()
   400→      b.x = a.x
   401→      b.y = a.y + 0.08 // feet rest on the seat board
   402→      // track tangential velocity so release preserves momentum
   403→      b.vx = sw.omega * sw.L * Math.cos(sw.theta)
   404→      b.vy = sw.omega * sw.L * Math.sin(sw.theta)
   405→      this.setPrompt('LET GO')
   406→      if (jump || interact) this.release()
   407→      // airy whoosh when sweeping through the bottom of the arc at speed
   408→      this.whooshT -= dt
   409→      if (Math.abs(sw.omega) > 3.1 && Math.abs(sw.theta) < 0.24 && this.whooshT <= 0) {
   410→        this.sfx.whoosh()
   411→        this.whooshT = 0.6
   412→      }
   413→      this.collectStars()
   414→      this.syncDuck(dt, pump / T.swingPump)
   415→      this.updateCamera(dt)
   416→      return
   417→    }
   418→
   419→    // ---- horizontal movement ----
   420→    const dir = this.input.dir()
   421→    if (dir !== 0) {
   422→      const target = dir * T.moveSpeed
   423→      const sameDir = Math.sign(b.vx) === dir
   424→      const over = Math.abs(b.vx) > T.moveSpeed + 0.05
   425→      if (over && sameDir) {
   426→        // preserve hard-earned momentum (e.g. swing releases); settle
   427→        // gently back to run speed only once grounded
   428→        if (b.onGround) b.vx = moveToward(b.vx, target, 8 * dt)
   429→      } else {
   430→        const rate = b.onGround ? T.groundAccel : T.airAccel
   431→        b.vx = moveToward(b.vx, target, rate * dt)
   432→      }
   433→    } else {
   434→      b.vx = moveToward(b.vx, 0, (b.onGround ? T.groundDecel : T.airDrag) * dt)
   435→    }
   436→
   437→    // unattended swings keep settling/swaying between grabs — released
   438→    // swings must actually lose energy, otherwise they spin forever and
   439→    // their seats never come back within grab range
   440→    for (const s of this.swings) s.settle(dt, this.time)
   441→
   442→    // ---- jump with coyote time + input buffer (no double jumps) ----
   443→    this.coyote = b.onGround ? T.coyote : Math.max(0, this.coyote - dt)
   444→    if (jump) this.jumpBuf = T.buffer
   445→    else this.jumpBuf = Math.max(0, this.jumpBuf - dt)
   446→    if (this.jumpBuf > 0 && this.coyote > 0 && b.vy <= 0.01) {
   447→      b.vy = T.jumpVel
   448→      this.jumpBuf = 0
   449→      this.coyote = 0
   450→      b.onGround = false
   451→      this.duck.onJump(true)
   452→      this.sfx.jump()
   453→    }
   454→
   455→    // ---- gravity + collide ----
   456→    const fallBefore = b.vy
   457→    b.vy = Math.max(T.maxFall, b.vy + T.gravity * dt)
   458→    const was = b.onGround
   459→    const res = this.world.move(b, dt, was)
   460→    if (res.landed) {
   461→      const fs = Math.abs(Math.min(0, fallBefore))
   462→      if (fs > 2.5) this.sfx.land()
   463→      this.duck.onLand(fs)
   464→    }
   465→    if (res.landed && this.checkCheckpoints()) {
   466→      // a fresh checkpoint activation gets a tiny celebration
   467→      const cp = this.checkpoints[this.cpIndex]
   468→      this.duck.celebrate()
   469→      this.sfx.chime()
   470→      if (cp) this.burst(cp.x, cp.y + 1.3, 12)
   471→      this.cb.onTip('checkpoint')
   472→      this.cb.onCheckpoint(this.cpIndex + 1, this.checkpoints.length)
   473→    }
   474→
   475→    // ---- footsteps follow the run cycle ----
   476→    if (b.onGround && Math.abs(b.vx) > 0.8) {
   477→      if (this.duck.consumeStep()) {
   478→        this.stepAlt = !this.stepAlt
   479→        this.sfx.step(this.stepAlt)
   480→      }
   481→    }
   482→
   483→    // ---- swing interaction: nearest seat within ITS grab range, preferring
   484→    // the swing the duck is moving toward (never grabs a seat behind its
   485→    // back when the next swing ahead is also in range) ----
   486→    const moveDir = Math.abs(b.vx) > 0.5 ? Math.sign(b.vx) : 0
   487→    let nearest: Swing | null = null
   488→    let nearScore = Infinity
   489→    for (const s of this.swings) {
   490→      const seat = s.seatAnchor()
   491→      const d = Math.hypot(b.x - seat.x, b.y + 0.35 - seat.y)
   492→      const wasInRange = s.inRangeNow
   493→      const inRange = d < s.range && !(s === this.regrabSwing && this.regrabT > 0)
   494→      s.setInRange(inRange)
   495→      // first moment a swing becomes reachable: a soft glint + tiny sparkle
   496→      // helps players read WHERE the next interaction lives (edge-triggered,
   497→      // cooled down so cruising past swings never turns into a siren)
   498→      if (inRange && !wasInRange && !this.swinging) {
   499→        const last = this.glintCool.get(s) ?? -9
   500→        if (this.time - last > 1.4) {
   501→          this.glintCool.set(s, this.time)
   502→          this.sfx.glint()
   503→          this.burst(seat.x, seat.y + 0.3, 4)
   504→        }
   505→      }
   506→      if (inRange) {
   507→        const ahead = moveDir !== 0 ? Math.max(0, moveDir * (seat.x - b.x)) : 0
   508→        const score = d - 0.45 * Math.min(1.2, ahead)
   509→        if (score < nearScore) {
   510→          nearScore = score
   511→          nearest = s
   512→        }
   513→      }
   514→    }
   515→    if (nearest) {
   516→      this.setPrompt('GRAB')
   517→      if (interact) this.grab(nearest)
   518→    } else {
   519→      this.setPrompt('')
   520→    }
   521→
   522→    this.collectStars()
   523→
   524→    // ---- safe recovery when falling out of the level ----
   525→    if (b.y < this.killY) this.respawn(true)
   526→
   527→    // ---- end flag ----
   528→    if (b.x > this.flagX - 0.35 && b.onGround) {
   529→      this.won = true
   530→      this.hopTimer = 0.3
   531→      this.sfx.win()
   532→      this.burst(b.x, b.y + 1.2, 26)
   533→      this.burst(b.x - 1.2, b.y + 2.2, 18)
   534→      this.burst(b.x + 1.2, b.y + 2.2, 18)
   535→      this.cb.onVictory(true, {
   536→        stars: this.starsGot,
   537→        starsTotal: this.level.stars.length,
   538→        levelIndex: this.levelIdx,
   539→        levelName: this.def.name,
   540→        time: this.levelTime,
   541→      })
   542→    }
   543→
   544→    this.syncDuck(dt, 0)
   545→    this.updateCamera(dt)
   546→  }
   547→
   548→  // ============================ stars ============================
   549→
   550→  private collectStars(): void {
   551→    const b = this.body
   552→    for (const s of this.level.stars) {
   553→      if (s.taken) continue
   554→      if (Math.hypot(b.x - s.x, b.y + 0.4 - s.y) < T.starRadius) {
   555→        s.taken = true
   556→        s.mesh.visible = false
   557→        this.starsGot++
   558→        this.sfx.collect()
   559→        this.burst(s.x, s.y, 10)
   560→        this.cb.onStars(this.starsGot, this.level.stars.length)
   561→        if (!this.tipSent.star) {
   562→          this.tipSent.star = true
   563→          this.cb.onTip('star')
   564→        }
   565→      }
   566→    }
   567→  }

   569→  // ============================ confetti (shared bursts) ============================
   570→
   571→  private burst(x: number, y: number, n: number): void {
   572→    for (let i = 0; i < n; i++) {
   573→      const m = new THREE.Mesh(
   574→        this.confettiGeo,
   575→        new THREE.MeshBasicMaterial({
   576→          color: this.confettiColors[i % this.confettiColors.length],
   577→          transparent: true,
   578→          opacity: 0.95,
   579→        }),
   580→      )
   581→      m.position.set(x + (Math.random() - 0.5) * 0.3, y + (Math.random() - 0.5) * 0.3, (Math.random() - 0.5) * 0.5)
   582→      m.scale.setScalar(0.7 + Math.random() * 0.9)
   583→      const a = Math.random() * Math.PI * 2
   584→      const sp = 1.6 + Math.random() * 2.6
   585→      this.confetti.push({
   586→        mesh: m,
   587→        vx: Math.cos(a) * sp,
   588→        vy: 2.4 + Math.random() * 3,
   589→        spin: (Math.random() - 0.5) * 16,
   590→        life: 0.75 + Math.random() * 0.4,
   591→        max: 1.15,
   592→        grow: 0.1,
   593→        gravity: 6.5,
   594→      })
   595→      this.scene.add(m)
   596→    }
   597→  }
   598→
   599→  private tickConfetti(dt: number): void {
   600→    for (let i = this.confetti.length - 1; i >= 0; i--) {
   601→      const p = this.confetti[i]
   602→      p.life -= dt
   603→      if (p.life <= 0) {
   604→        this.scene.remove(p.mesh)
   605→        ;(p.mesh.material as THREE.Material).dispose()
   606→        this.confetti.splice(i, 1)
   607→        continue
   608→      }
   609→      p.vy -= p.gravity * dt
   610→      p.vx *= 1 - 0.9 * dt
   611→      p.mesh.position.x += p.vx * dt
   612→      p.mesh.position.y += p.vy * dt
   613→      p.mesh.rotation.x += p.spin * dt
   614→      p.mesh.rotation.y += p.spin * 0.6 * dt
   615→      ;(p.mesh.material as THREE.MeshBasicMaterial).opacity = Math.min(1, p.life / 0.4)
   616→    }
   617→  }

   619→  // ============================ duck sync ============================
   620→
   621→  private syncDuck(dt: number, pump = 0): void {
   622→    const b = this.body
   623→    if (this.swinging && this.activeSwing) {
   624→      if (Math.abs(this.activeSwing.omega) > 0.25) this.duck.facing = this.activeSwing.omega > 0 ? 1 : -1
   625→    } else if (b.vx > 0.25) {
   626→      this.duck.facing = 1
   627→    } else if (b.vx < -0.25) {
   628→      this.duck.facing = -1
   629→    }
   630→    this.duck.root.position.set(b.x, b.y, 0)
   631→    // highest surface below the duck — feeds the contact shadow and
   632→    // gives the player a readable landing cue over gaps (movers included)
   633→    let groundY = -8
   634→    const consider = (x1: number, x2: number, top: number) => {
   635→      if (b.x >= x1 - 0.25 && b.x <= x2 + 0.25 && top <= b.y - b.hh + 0.08 && top > groundY) groundY = top
   636→    }
   637→    for (const p of this.plats) consider(p.x1, p.x2, p.top)
   638→    for (const box of this.moverBoxes) consider(box.cx - box.hw, box.cx + box.hw, box.cy + box.hh)
   639→    const speed = Math.hypot(b.vx, b.vy)
   640→    this.duck.update(dt, {
   641→      vx: b.vx,
   642→      vy: b.vy,
   643→      grounded: b.onGround,
   644→      swinging: this.swinging,
   645→      falling: !b.onGround && b.vy < -3,
   646→      fast: !b.onGround && !this.swinging && speed > 6.4,
   647→      tilt: this.swinging && this.activeSwing ? this.activeSwing.theta : 0,
   648→      pump,
   649→      won: this.won,
   650→      groundY,
   651→    })
   652→    this.dirLight.position.set(b.x + 5, 11, 8)
   653→    this.dirLight.target.position.set(b.x, 0, 0)
   654→  }
   655→
   656→  private updateCamera(dt: number): void {
   657→    const b = this.body
   658→    // look-ahead follows SMOOTHED velocity: anticipates launches and fast
   659→    // flights without ever snapping or oscillating
   660→    this.smVx += (b.vx - this.smVx) * Math.min(1, dt * 3.2)
   661→    const airborneFast = !b.onGround && !this.swinging && Math.abs(this.smVx) > 6
   662→    let lead = clamp(this.smVx * (airborneFast ? 0.62 : 0.45), -3.4, 3.4)
   663→    // while swinging, lean the frame into the arc's travel direction so the
   664→    // landing zone stays readable through the widest part of the sweep
   665→    if (this.swinging && this.activeSwing) {
   666→      lead += clamp(this.activeSwing.omega * this.activeSwing.L * 0.075, -1.1, 1.1)
   667→    }
   668→    const look = lead + this.duck.facing * 0.35
   669→    const tx = b.x + look
   670→    const ty = Math.max(0.5, b.y * 0.5 + 1.0) + (this.swinging ? 0.4 : 0)
   671→    const kx = 1 - Math.exp(-3.6 * dt)
   672→    const ky = 1 - Math.exp(-3 * dt)
   673→    this.camX += (tx - this.camX) * kx
   674→    this.camY += (ty - this.camY) * ky
   675→    // breathe out slightly while swinging to reveal the landing zone
   676→    const tz = this.camZ * (this.swinging ? 1.12 : 1)
   677→    this.camZCur += (tz - this.camZCur) * Math.min(1, dt * 2.5)
   678→    this.camera.position.set(this.camX, this.camY + 1.5, this.camZCur)
   679→    this.camera.lookAt(this.camX + 0.3, this.camY + 0.35, 0)
   680→  }

   682→  // ============================ swing chain ============================
   683→
   684→  private grab(sw: Swing): void {
   685→    const b = this.body
   686→    sw.theta = clamp(sw.angleFrom(b.x, b.y + 0.35), -1.2, 1.2)
   687→    sw.omega = clamp(b.vx / sw.L, -2.6, 2.6)
   688→    this.activeSwing = sw
   689→    this.swinging = true
   690→    b.vx = 0
   691→    b.vy = 0
   692→    b.onGround = false
   693→    for (const s of this.swings) s.setInRange(false)
   694→    sw.bump()
   695→    this.duck.onGrab()
   696→    this.sfx.grab()
   697→    this.sfx.sparkle()
   698→    if (!this.tipSent.pump) {
   699→      this.tipSent.pump = true
   700→      this.cb.onTip('pump')
   701→    }
   702→  }
   703→
   704→  private release(): void {
   705→    const sw = this.activeSwing
   706→    if (!sw) return
   707→    const v = sw.releaseVelocity()
   708→    const b = this.body
   709→    b.vx = v.x * T.releaseBoost
   710→    b.vy = Math.max(v.y * T.releaseBoost, -2)
   711→    // the duck kicks off the seat when releasing on a forward swing —
   712→    // a small, readable boost that rewards good timing
   713→    if (sw.omega > 0.3) b.vy += T.forwardKick
   714→    if (Math.abs(b.vx) < 2.4) {
   715→      b.vx = (sw.omega >= 0 ? 1 : -1) * 2.4
   716→    }
   717→    this.swinging = false
   718→    this.activeSwing = null
   719→    // brief cooldown so flying past THIS swing doesn't instantly re-grab it;
   720→    // any other swing in range stays grabbable for mid-air chains. The base
   721→    // cooldown scales with release speed: a hard launch leaves the old
   722→    // swing's grab zone later, so it must stay locked longer (a weak drop
   723→    // only locks it for a blink, keeping low-release re-grabs snappy).
   724→    this.regrabSwing = sw
   725→    this.regrabT = T.regrabCooldown + Math.min(0.55, Math.hypot(b.vx, b.vy) * 0.05)
   726→    b.onGround = false
   727→    this.duck.onJump()
   728→    this.duck.onRelease()
   729→    this.sfx.release()
   730→  }

   732→  // ============================ checkpoints / lifecycle ============================
   733→
   734→  /** Activate the next checkpoint when the duck lands on its platform. */
   735→  private checkCheckpoints(): boolean {
   736→    const b = this.body
   737→    for (let i = this.cpIndex + 1; i < this.checkpoints.length; i++) {
   738→      const cp = this.checkpoints[i]
   739→      const p = this.plats[cp.plat]
   740→      if (!p) continue
   741→      if (b.x > p.x1 + 0.3 && b.x < p.x2 - 0.3 && Math.abs(b.y - b.hh - cp.y) < 0.4) {
   742→        this.cpIndex = i
   743→        this.level?.setCheckpointActive(i)
   744→        return true
   745→      }
   746→    }
   747→    return false
   748→  }
   749→
   750→  respawn(withFlash: boolean): void {
   751→    const cp = this.cpIndex >= 0 ? this.checkpoints[this.cpIndex] : { x: this.spawnX, y: 0 }
   752→    const b = this.body
   753→    b.x = cp.x
   754→    b.y = cp.y + T.colliderH / 2 + 0.02
   755→    b.vx = 0
   756→    b.vy = 0
   757→    b.onGround = false
   758→    b.carrier = null
   759→    this.swinging = false
   760→    this.activeSwing = null
   761→    this.regrabSwing = null
   762→    this.regrabT = 0
   763→    this.debugRelease = null
   764→    for (const s of this.swings) s.setInRange(false)
   765→    this.duck.facing = 1
   766→    this.duck.dizzy()
   767→    this.camX = b.x
   768→    this.camY = Math.max(1.0, b.y)
   769→    this.sfx.fall()
   770→    if (withFlash) this.cb.onFlash()
   771→  }
   772→
   773→  reset(): void {
   774→    if (this.won) {
   775→      this.won = false
   776→      this.cb.onVictory(false)
   777→    }
   778→    this.hopTimer = 0
   779→    this.cpIndex = -1
   780→    this.starsGot = 0
   781→    this.levelTime = 0
   782→    this.tipSent = { pump: false, star: false }
   783→    for (const s of this.level.stars) {
   784→      s.taken = false
   785→      s.mesh.visible = true
   786→    }
   787→    this.cb.onStars(0, this.level.stars.length)
   788→    for (const s of this.swings) {
   789→      s.theta = 0
   790→      s.omega = 0
   791→    }
   792→    this.respawn(false)
   793→  }
   794→
   795→  /** Quick retry: back to the latest checkpoint, stars kept. */
   796→  retry(): void {
   797→    if (this.won) {
   798→      this.won = false
   799→      this.cb.onVictory(false)
   800→    }
   801→    this.respawn(true)
   802→  }

   804→  /** Full restart from the very beginning (PLAY AGAIN). */
   805→  fullReset(): void {
   806→    this.reset()
   807→  }

   842→  private key(e: KeyboardEvent, down: boolean): void {
   843→    if (e.repeat) return
   844→    const c = e.code
   845→    if (c === 'ArrowLeft' || c === 'ArrowRight' || c === 'ArrowUp' || c === 'Space') e.preventDefault()
   846→    if (down) this.sfx.unlock()
   847→    if (c === 'ArrowLeft' || c === 'KeyA') this.input.set('left', down)
   848→    else if (c === 'ArrowRight' || c === 'KeyD') this.input.set('right', down)
   849→    else if (c === 'Space' || c === 'ArrowUp' || c === 'KeyW') this.input.set('jump', down)
   850→    else if (c === 'KeyE' || c === 'Enter') this.input.set('interact', down)
   851→    else if (c === 'KeyR' && down) this.retry()
   852→  }
